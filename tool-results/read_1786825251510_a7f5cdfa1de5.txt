  1707→    /* ============================
  1708→       STORIES CAROUSEL
  1709→       With drag, swipe, rubber-band
  1710→       ============================ */
  1711→    class StoriesCarousel {
  1712→      constructor(track, opts = {}) {
  1713→        this.track = track;
  1714→        this.cards = Array.from(track.children);
  1715→        this.currentIndex = 0;
  1716→        this.isDown = false;
  1717→        this.startX = 0;
  1718→        this.startOffset = 0;
  1719→        this.currentX = 0;
  1720→        this.velocity = 0;
  1721→        this.lastX = 0;
  1722→        this.lastTime = 0;
  1723→        this.cardWidth = 0;
  1724→        this.maxScroll = 0;
  1725→        this.autoAdvance = opts.autoAdvance !== false;
  1726→        this.autoAdvanceInterval = opts.interval || 5000;
  1727→        this.autoTimer = null;
  1728→        this.pauseAuto = false;
  1729→        
  1730→        this.init();
  1731→      }
  1732→      
  1733→      init() {
  1734→        this.calculateDimensions();
  1735→        this.bindEvents();
  1736→        this.buildDots();
  1737→        this.startAuto();
  1738→        
  1739→        window.addEventListener('resize', () => {
  1740→          this.calculateDimensions();
  1741→          this.clamp();
  1742→          this.applyTransform(0);
  1743→        });
  1744→      }
  1745→      
  1746→      calculateDimensions() {
  1747→        if (this.cards.length === 0) return;
  1748→        const trackStyle = window.getComputedStyle(this.track);
  1749→        const gap = parseFloat(trackStyle.gap) || 24;
  1750→        const paddingLeft = parseFloat(trackStyle.paddingLeft) || 32;
  1751→        this.cardWidth = this.cards[0].offsetWidth + gap;
  1752→        const containerWidth = this.track.parentElement.offsetWidth;
  1753→        const totalWidth = (this.cards.length * this.cardWidth) - gap + paddingLeft * 2;
  1754→        this.maxScroll = Math.max(0, totalWidth - containerWidth);
  1755→      }
  1756→      
  1757→      bindEvents() {
  1758→        const down = (e) => this.handleDown(e);
  1759→        const move = (e) => this.handleMove(e);
  1760→        const up = () => this.handleUp();
  1761→        
  1762→        this.track.addEventListener('mousedown', down);
  1763→        this.track.addEventListener('touchstart', down, { passive: true });
  1764→        window.addEventListener('mousemove', move);
  1765→        window.addEventListener('touchmove', move, { passive: false });
  1766→        window.addEventListener('mouseup', up);
  1767→        window.addEventListener('touchend', up);
  1768→        window.addEventListener('touchcancel', up);
  1769→        
  1770→        // Pause auto on hover
  1771→        this.track.addEventListener('mouseenter', () => this.pauseAuto = true);
  1772→        this.track.addEventListener('mouseleave', () => this.pauseAuto = false);
  1773→        
  1774→        // Prev/Next buttons
  1775→        const prev = document.getElementById('storyPrev');
  1776→        const next = document.getElementById('storyNext');
  1777→        if (prev) prev.addEventListener('click', () => this.goTo(this.currentIndex - 1));
  1778→        if (next) next.addEventListener('click', () => this.goTo(this.currentIndex + 1));
  1779→      }
  1780→      
  1781→      handleDown(e) {
  1782→        this.isDown = true;
  1783→        this.track.classList.add('dragging');
  1784→        const x = e.type.includes('touch') ? e.touches[0].pageX : e.pageX;
  1785→        this.startX = x;
  1786→        this.startOffset = this.currentX;
  1787→        this.lastX = x;
  1788→        this.lastTime = Date.now();
  1789→        this.velocity = 0;
  1790→        this.stopAuto();
  1791→      }
  1792→      
  1793→      handleMove(e) {
  1794→        if (!this.isDown) return;
  1795→        if (e.cancelable) e.preventDefault();
  1796→        const x = e.type.includes('touch') ? e.touches[0].pageX : e.pageX;
  1797→        const delta = x - this.startX;
  1798→        let newX = this.startOffset + delta;
  1799→        
  1800→        // Rubber band at edges
  1801→        if (newX > 0) {
  1802→          newX = newX * 0.35;
  1803→        } else if (newX < -this.maxScroll) {
  1804→          const over = newX + this.maxScroll;
  1805→          newX = -this.maxScroll + over * 0.35;
  1806→        }
  1807→        
  1808→        // Track velocity
  1809→        const now = Date.now();
  1810→        const dt = now - this.lastTime;
  1811→        if (dt > 0) {
  1812→          this.velocity = (x - this.lastX) / dt;
  1813→        }
  1814→        this.lastX = x;
  1815→        this.lastTime = now;
  1816→        
  1817→        this.currentX = newX;
  1818→        this.applyTransform(0);
  1819→      }
  1820→      
  1821→      handleUp() {
  1822→        if (!this.isDown) return;
  1823→        this.isDown = false;
  1824→        this.track.classList.remove('dragging');
  1825→        
  1826→        // Apply momentum
  1827→        let target = this.currentX + this.velocity * 300;
  1828→        
  1829→        // Snap to nearest card
  1830→        const snapIndex = Math.round(Math.abs(target) / this.cardWidth);
  1831→        const clampedIndex = Math.max(0, Math.min(this.cards.length - 1, snapIndex));
  1832→        let snappedX = -clampedIndex * this.cardWidth;
  1833→        
  1834→        // Clamp
  1835→        if (snappedX > 0) snappedX = 0;
  1836→        if (snappedX < -this.maxScroll) snappedX = -this.maxScroll;
  1837→        
  1838→        this.currentX = snappedX;
  1839→        this.currentIndex = clampedIndex;
  1840→        this.applyTransform(700);
  1841→        this.updateDots();
  1842→        
  1843→        // Resume auto after delay
  1844→        setTimeout(() => this.startAuto(), 1500);
  1845→      }
  1846→      
  1847→      goTo(idx) {
  1848→        this.currentIndex = Math.max(0, Math.min(this.cards.length - 1, idx));
  1849→        let target = -this.currentIndex * this.cardWidth;
  1850→        if (target > 0) target = 0;
  1851→        if (target < -this.maxScroll) target = -this.maxScroll;
  1852→        this.currentX = target;
  1853→        this.applyTransform(700);
  1854→        this.updateDots();
  1855→        this.stopAuto();
  1856→        setTimeout(() => this.startAuto(), 2000);
  1857→      }
  1858→      
  1859→      clamp() {
  1860→        if (this.currentX > 0) this.currentX = 0;
  1861→        if (this.currentX < -this.maxScroll) this.currentX = -this.maxScroll;
  1862→      }
  1863→      
  1864→      applyTransform(duration = 0) {
  1865→        if (duration > 0) {
  1866→          this.track.style.transition = `transform ${duration}ms cubic-bezier(0.16, 1, 0.3, 1)`;
  1867→        } else {
  1868→          this.track.style.transition = 'none';
  1869→        }
  1870→        this.track.style.transform = `translateX(${this.currentX}px)`;
  1871→      }
  1872→      
  1873→      buildDots() {
  1874→        const dotsContainer = document.getElementById('storyDots');
  1875→        if (!dotsContainer) return;
  1876→        dotsContainer.innerHTML = '';
  1877→        this.cards.forEach((_, i) => {
  1878→          const dot = document.createElement('button');
  1879→          dot.className = 'story-dot transition-all';
  1880→          dot.style.cssText = i === 0 
  1881→            ? 'width:32px;height:2px;background:var(--accent);'
  1882→            : 'width:8px;height:2px;background:var(--border-light);';
  1883→          dot.addEventListener('click', () => this.goTo(i));
  1884→          dotsContainer.appendChild(dot);
  1885→        });
  1886→      }
  1887→      
  1888→      updateDots() {
  1889→        const dots = document.querySelectorAll('#storyDots button');
  1890→        dots.forEach((dot, i) => {
  1891→          dot.style.cssText = i === this.currentIndex 
  1892→            ? 'width:32px;height:2px;background:var(--accent);transition:all 0.4s;'
  1893→            : 'width:8px;height:2px;background:var(--border-light);transition:all 0.4s;';
  1894→        });
  1895→      }
  1896→      
  1897→      startAuto() {
  1898→        if (!this.autoAdvance) return;
  1899→        this.stopAuto();
  1900→        this.autoTimer = setInterval(() => {
  1901→          if (this.pauseAuto || this.isDown) return;
  1902→          let next = this.currentIndex + 1;
  1903→          if (next >= this.cards.length) next = 0;
  1904→          this.goTo(next);
  1905→        }, this.autoAdvanceInterval);
  1906→      }
  1907→      
  1908→      stopAuto() {
  1909→        if (this.autoTimer) {
  1910→          clearInterval(this.autoTimer);
  1911→          this.autoTimer = null;
  1912→        }
  1913→      }
  1914→    }
  1915→    
  1916→    new StoriesCarousel(document.getElementById('storyTrack'), { interval: 4500 });
